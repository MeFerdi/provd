All endpoints live under `/api/v1` and return JSON. Timestamps are ISO 8601 in UTC. Addresses are 0x-prefixed hex.

## Endpoints

| Method | Path | Returns |
|---|---|---|
| `GET` | `/api/v1/protocols` | `ProtocolList` |
| `GET` | `/api/v1/protocols/{slug}` | `ProtocolDetail` |
| `POST` | `/api/v1/scan` | `Posture` |
| `POST` | `/api/v1/early-access` | `201` with `{"ok": true}` |

### POST /scan

Request body:

```json
{ "chain": "ethereum", "address": "0x…40 hex characters" }
```

`chain` is `ethereum` today.

### POST /early-access

Request body:

```json
{
  "email": "you@example.com",
  "organization": "optional",
  "role": "curator | insurer | protocol | fund | researcher | other",
  "message": "optional"
}
```

## Errors

| Status | Meaning |
|---|---|
| `404` | Unknown protocol slug. |
| `422` | Invalid input, for example a malformed address or a contract that is not a LayerZero OApp. The `detail` field explains why. |
| `429` | Rate limited. Retry later with backoff. |
| `502` / `504` | The upstream RPC could not be reached or timed out. Retry later. |

Error bodies have the form `{"detail": "…"}`.

## Types

```ts
type Severity = "critical" | "high" | "medium" | "low" | "info";
type RiskLevel = "critical" | "high" | "medium" | "low";
type Grade = "A" | "B" | "C" | "D" | "F";

interface Verifier { address: string; name: string | null }

interface UlnSide {
  library: string;
  is_default_library: boolean;
  confirmations: number;
  required_dvns: Verifier[];
  optional_dvns: Verifier[];
  optional_threshold: number;
  effective_verifiers: number;
}

interface Pathway {
  eid: number;               // LayerZero endpoint id of the remote chain
  chain_name: string | null;
  peer: string;
  receive: UlnSide | null;
  send: UlnSide | null;
}

interface AdminAccount {
  address: string;
  type: "eoa" | "safe" | "contract" | "none" | "unknown";
  threshold: number | null;  // Safe only
  owners: number | null;     // Safe only
}

interface Admin { owner: AdminAccount | null; delegate: AdminAccount | null }

interface Evidence { label: string; value: string; link: string | null }

interface Finding {
  rule_id: string;
  severity: Severity;
  title: string;
  summary: string;
  why_it_matters: string;
  recommendation: string;
  pathways: number[];        // affected eids
  evidence: Evidence[];
}

interface FindingRef { rule_id: string; severity: Severity; title: string }

interface ChangeEvent {
  detected_at: string;
  kind: string;              // e.g. CHG-VERIFIER-REDUCED
  severity: Severity;
  summary: string;
  pathway: number | null;
  before: object | null;
  after: object | null;
}

interface Posture {
  framework: "layerzero_v2";
  chain: string;
  contract: string;
  grade: Grade;
  score: number;
  risk_level: RiskLevel;
  pathways_total: number;
  single_verifier_pathways: number;
  admin: Admin;
  pathways: Pathway[];
  findings: Finding[];
  checked_at: string;
}

interface ProtocolSummary {
  slug: string;
  name: string;
  category: string;
  framework: string;
  chain: string;
  contract: string;
  grade: Grade;
  score: number;
  risk_level: RiskLevel;
  pathways_total: number;
  single_verifier_pathways: number;
  top_findings: FindingRef[];
  checked_at: string;
  last_change_at: string | null;
}

interface ProtocolDetail extends ProtocolSummary {
  website: string | null;
  admin: Admin;
  pathways: Pathway[];
  findings: Finding[];
  changes: ChangeEvent[];
}

interface ProtocolList { generated_at: string; items: ProtocolSummary[] }
```

## Field notes

- `effective_verifiers` is the number of independent verifier attestations a message needs on that side of the route: required DVNs plus the optional threshold. A value of `1` is what the `LZ-SINGLE-VERIFIER` rule flags.
- `is_default_library` is `true` when the route inherits LayerZero's default configuration rather than an explicit one.
- `rule_id` values and their severities are listed in the [methodology](/methodology).
